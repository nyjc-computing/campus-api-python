    Campus API (campus.api)¶

    The /api/v1 service: assignments, circles, files, hosted apps, submissions, and timetable. Address collections directly (campus.api.assignments.list()), single items by id (campus.api.circles["circle_ab12cd34"].get()), and drill into sub-resources (... submissions["sub_1"].feedback.add(...)).

    All methods raise the mapped APIError subclass on non-2xx responses. Records deserialize to campus.model classes, except files, which are plain validated dicts.

    Note: campus.model.* types are documented in the campus repo; signatures here reference them without cross-links.

    class Assignments(ResourceCollection):
    """Campus API Assignments resource."""
    path = "assignments"

    def __getitem__(self, assignment_id: str) -> "Assignments.Assignment":
        """Get a specific assignment resource by ID."""
        return Assignments.Assignment(assignment_id, parent=self)

    def list(self, *, created_by: "str | None" = None) -> "list[campus.model.Assignment]":
        """List all assignments matching filter requirements.

        Args:
            created_by: Filter by teacher who created the assignment

        Returns:
            List of Assignment objects
        """
        params = {}
        if created_by:
            params["created_by"] = created_by

        body = self._get_json(self.make_path(), query=params)
        return [
            campus.model.Assignment.from_resource(item)
            for item in body["data"]
        ]

    def new(
            self,
            *,
            title: str,
            description: "str | None" = None,
            questions: "list[dict] | None" = None,
            classroom_links: "list[dict] | None" = None,
    ) -> campus.model.Assignment:
        """Create a new assignment.

        Args:
            title: Assignment title (required by the server).
            description: Assignment description.
            questions: Question definitions.
            classroom_links: Google Classroom course/coursework links.

        Returns:
            The created Assignment.
        """
        return campus.model.Assignment.from_resource(self._post_json(
            self.make_path(),
            json=payload(
                title=title,
                description=description,
                questions=questions,
                classroom_links=classroom_links,
            )
        ))

    class Assignment(Resource):
        """Single campus API assignment resource."""

        @property
        def links(self) -> "Assignments.Assignment.Links":
            """Get the links resource for this assignment."""
            return Assignments.Assignment.Links("links", parent=self)

        def delete(self) -> None:
            """Delete this assignment."""
            self._delete_json(self.make_path(end_slash=True))

        def get(self) -> campus.model.Assignment:
            """Get this assignment.

            Returns:
                The Assignment.
            """
            body = self._get_json(self.make_path(end_slash=True))
            return campus.model.Assignment.from_resource(body)

        def update(self, **fields) -> None:
            """Update this assignment with the given fields.

            The server rejects an empty PATCH body, so at least one
            field must be provided.
            """
            self._patch_json(self.make_path(end_slash=True), json=updates(**fields))
            return None

        class Links(Resource):

        """Get a specific circle resource by ID."""

    def new(

        Args:

    class Circle(Resource):
        """Single campus API circle resource."""

        @property
        def members(self) -> "Circles.Circle.CircleMembers":
            """Get the members resource for this circle."""
            return Circles.Circle.CircleMembers("members", parent=self)

        def delete(self) -> None:
            """Delete this circle."""
            self._delete_json(self.make_path(end_slash=True))

        def get(self) -> campus.model.Circle:
            """Get this circle.

            Returns:
                The Circle.
            """
            body = self._get_json(self.make_path(end_slash=True))
            return campus.model.Circle.from_resource(body)

        def update(
                self,
                *,
                name: str,
                description: "str | None" = None,
        ) -> None:
            """Update this circle.

            The server requires name on every circle PATCH and returns
            422 without it, so it is a required parameter here.
            Description is unchanged when omitted.

            Args:
                name: New circle name (must not collide with an
                    existing circle's name)
                description: New description; omitted leaves it unchanged
            """
            payload: dict[str, Any] = {"name": name}
            if description is not None:
                payload["description"] = description
            self._patch_json(self.make_path(end_slash=True), json=payload)
            return None

        class CircleMembers(Resource):
            """Campus API Circle Members resource."""

            def list(self) -> dict[str, int]:
                """Return the circle's {member_id: access_value} mapping."""
                return self._get_json(self.make_path(end_slash=True))

            def add(self, member_id: str, access_value: int) -> None:
                """Add a member with an access value.

                Targets the /members/add action route; adding an
                existing member re-errors server-side. Use set() for
                upsert semantics.
                """
                self._post_json(
                    self.make_path("add"),
                    json={"member_id": member_id, "access_value": access_value}
                )

            def remove(self, member_id: str) -> None:
                """Remove a member.

                Targets the /members/remove action route.
                """
                self._delete_json(
                    self.make_path("remove"),
                    json={"member_id": member_id}
                )

            def set(self, member_id: str, access_value: int) -> None:
                """Create or update a member's access value (upsert).

                Unlike add()/remove(), which target the /members/add and
                /members/remove action routes, this PATCHes the members
                collection directly — the server's set semantics apply
                (no error when the access value is unchanged).
                """
                self._patch_json(
                    self.make_path(end_slash=True),
                    json={"member_id": member_id, "access_value": access_value}
                )
                return None
